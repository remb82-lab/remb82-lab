function renderDiagonals(){const r=room(),pts=geometrySummary(r).points,ds=r.geometry.diagonalMeasurements||[];$('diagonalList').innerHTML=ds.length?ds.map((d,i)=>{const c=diagonalLength(pts,d.a,d.b),delta=d.length?Math.abs(c-num(d.length)):0;return `<div class="data-row"><div><b>${String.fromCharCode(65+d.a)}–${String.fromCharCode(65+d.b)}</b><small>расчёт ${fmt(c)} м${d.length?` · Δ ${fmt(delta)} м`:''}</small><div class="row-fields"><label>От<input data-da="${i}" type="number" min="1" step="1" value="${d.a+1}"></label><label>До<input data-db="${i}" type="number" min="1" step="1" value="${d.b+1}"></label><label>Замер, м<input data-dl="${i}" type="number" min="0" step="0.01" value="${d.length||''}"></label></div></div><button class="remove" data-ddel="${i}">×</button></div>`}).join(''):'<div class="hint">Контрольных диагоналей нет.</div>';qa('[data-da]').forEach(e=>e.onchange=()=>{ds[+e.dataset.da].a=Math.max(0,Math.min(pts.length-1,Math.round(num(e.value))-1));persist();renderDiagonals();renderPlan()});qa('[data-db]').forEach(e=>e.onchange=()=>{ds[+e.dataset.db].b=Math.max(0,Math.min(pts.length-1,Math.round(num(e.value))-1));persist();renderDiagonals();renderPlan()});qa('[data-dl]').forEach(e=>e.oninput=()=>{ds[+e.dataset.dl].length=Math.max(0,num(e.value));persist();renderPlan()});qa('[data-ddel]').forEach(e=>e.onclick=()=>{ds.splice(+e.dataset.ddel,1);persist('Диагональ удалена');renderDiagonals();renderPlan()})}
